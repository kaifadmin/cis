# Title: First Order Logic using Family Relationship Facts
father = {
    "Raj": "Amit",
    "Raj": "Neha"
}

mother = {
    "Priya": "Amit",
    "Priya": "Neha"
}

parents = [
    ("Raj", "Amit"),
    ("Priya", "Amit"),
    ("Raj", "Neha"),
    ("Priya", "Neha")
]

print("Parent Relationships:")
for p, c in parents:
    print(p, "is parent of", c)
print()

def is_parent(p, c):
    return (p, c) in parents

def is_sibling(c1, c2):
    if c1 == c2:
        return False
    p1 = [p for p, c in parents if c == c1]
    p2 = [p for p, c in parents if c == c2]
    return p1 == p2

print("Is Raj parent of Amit?")
print(is_parent("Raj", "Amit"))
print()
print("Are Amit and Neha siblings?")
print(is_sibling("Amit", "Neha"))

# Practical 6
# Title: A* Search Algorithm
import heapq

graph = {
    'A': [('B', 1), ('C', 3)],
    'B': [('D', 3), ('E', 6)],
    'C': [('F', 5)],
    'D': [('G', 2)],
    'E': [('G', 1)],
    'F': [('G', 2)],
    'G': []
}

h = {
    'A': 6,
    'B': 4,
    'C': 4,
    'D': 2,
    'E': 1,
    'F': 2,
    'G': 0
}

def a_star(start, goal):
    q = [(0, start, [])]
    vis = set()
    while q:
        cost, node, path = heapq.heappop(q)
        if node in vis:
            continue
        path = path + [node]
        vis.add(node)
        if node == goal:
            return path
        for nbr, wt in graph[node]:
            if nbr not in vis:
                tot = cost + wt + h[nbr]
                heapq.heappush(q, (tot, nbr, path))
    return None

start = 'A'
goal = 'G'
res = a_star(start, goal)
print("Shortest Path :", " -> ".join(res))

# Practical 8
# Title: Tic-Tac-Toe using AI Decision-Making
board = [" " for _ in range(9)]

def display_board():
    print()
    print(board[0], "|", board[1], "|", board[2])
    print("--+---+--")
    print(board[3], "|", board[4], "|", board[5])
    print("--+---+--")
    print(board[6], "|", board[7], "|", board[8])
    print()

def check_winner(p):
    wins = [
        [0, 1, 2], [3, 4, 5], [6, 7, 8],
        [0, 3, 6], [1, 4, 7], [2, 5, 8],
        [0, 4, 8], [2, 4, 6]
    ]
    for pos in wins:
        if all(board[i] == p for i in pos):
            return True
    return False

print("Welcome to Tic-Tac-Toe")
display_board()

for turn in range(9):
    if turn % 2 == 0:
        move = int(input("Enter position (1-9): ")) - 1
        if board[move] == " ":
            board[move] = "X"
        else:
            print("Position already occupied.")
            continue
    else:
        print("Computer's Turn")
        for i in range(9):
            if board[i] == " ":
                board[i] = "O"
                break
    
    display_board()

    if check_winner("X"):
        print("Player Wins!")
        break
    if check_winner("O"):
        print("Computer Wins!")
        break
else:
    print("Game Draw!")

import string

text = input("Enter a paragraph:\n")
text = text.lower()

for sym in string.punctuation:
    text = text.replace(sym, "")

words = text.split()
freq = {}

for w in words:
    if w in freq:
        freq[w] += 1
    else:
        freq[w] = 1

print("\nWord Frequency Analysis\n")

    print(w, ":", cnt)

# Title: AI Ethics and Fairness Analysis
data = [
    {"Name": "A1", "Gender": "Male", "Selected": "Yes"},
    {"Name": "A2", "Gender": "Female", "Selected": "Yes"},
    {"Name": "A3", "Gender": "Male", "Selected": "No"},
    {"Name": "A4", "Gender": "Female", "Selected": "Yes"},
    {"Name": "A5", "Gender": "Male", "Selected": "Yes"},
    {"Name": "A6", "Gender": "Female", "Selected": "No"}
]

m_tot = 0
f_tot = 0
m_sel = 0
f_sel = 0

for row in data:
    if row["Gender"] == "Male":
        m_tot += 1
        if row["Selected"] == "Yes":
            m_sel += 1
    else:
        f_tot += 1
        if row["Selected"] == "Yes":
            f_sel += 1

m_pct = (m_sel / m_tot) * 100
f_pct = (f_sel / f_tot) * 100

print("===== Fairness Analysis =====")
print("Male Selection Percentage :", m_pct, "%")
print("Female Selection Percentage :", f_pct, "%")

if m_pct == f_pct:
    print("\nResult : AI system appears Fair.")
else:
    print("\nResult : AI system may contain Bias.")
